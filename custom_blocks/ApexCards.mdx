import { useState } from "react";

export const ApexCards = ({
  items = [],
  containerClasses = "",
  gridColsMd = 2,
  layout = "default"
}) => {
  const [expandedCodeBlocks, setExpandedCodeBlocks] = useState({});
  const [copiedBlockKey, setCopiedBlockKey] = useState(null);
  const escapeHtml = (value = "") =>
    String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#39;");

  const markdownToHtml = (markdown = "") => {
    const codeSpans = [];
    let html = escapeHtml(markdown);

    html = html.replace(/`([^`]+)`/g, (_, code) => {
      const token = `__APEX_CODE_${codeSpans.length}__`;
      codeSpans.push(
        `<code class=\"rounded bg-base-300 px-1.5 py-0.5 text-sm font-mono\">${escapeHtml(
          code
        )}</code>`
      );
      return token;
    });

    html = html.replace(/\*\*(.+?)\*\*/g, "<strong>$1</strong>");
    html = html.replace(/\*(.+?)\*/g, "<em>$1</em>");
    html = html.replace(
      /\[([^\]]+)\]\((https?:\/\/[^\s)]+|\/[^\s)]+)\)/g,
      '<a href="$2" class="link link-hover">$1</a>'
    );
    html = html.replace(/\n/g, "<br />");

    codeSpans.forEach((snippet, index) => {
      html = html.replace(`__APEX_CODE_${index}__`, snippet);
    });

    return html;
  };

  const normalizeToneClasses = (iconClass = "") =>
    String(iconClass)
      .replace(/\btext-(warning|success|error|info)\b/g, "")
      .replace(/\s+/g, " ")
      .trim();

  const colsClass =
    gridColsMd === 4
      ? "md:grid-cols-4"
      : gridColsMd === 3
      ? "md:grid-cols-3"
      : gridColsMd === 2
      ? "md:grid-cols-2"
      : "md:grid-cols-1";

  const CardButton = ({ button, centered = false }) => {
    if (!button || !button.text) return null;

    const btnClass = [
      "btn btn-primary mt-4 inline-flex items-center gap-2 px-5 rounded-md font-semibold",
      centered ? "mx-auto" : ""
    ]
      .join(" ")
      .trim();

    if (button.href) {
      return (
        <a
          className={btnClass}
          href={button.href}
          target={button.target || "_self"}
          rel={button.target === "_blank" ? "noopener noreferrer" : undefined}
        >
          {button.text}
        </a>
      );
    }

    return (
      <button
        className={btnClass}
        type="button"
        onClick={button.onClick || (() => {})}
      >
        {button.text}
      </button>
    );
  };

  const RichDescription = ({ description, className = "" }) => {
    if (description == null) return null;

    if (typeof description !== "string") {
      return <div className={className}>{description}</div>;
    }

    return (
      <div
        className={className}
        dangerouslySetInnerHTML={{ __html: markdownToHtml(description) }}
      />
    );
  };

  const QAContent = ({ item }) => {
    const hasQuestion = item.question != null && String(item.question).trim() !== "";
    const hasResponse = item.response != null && String(item.response).trim() !== "";

    if (!hasQuestion && !hasResponse) return null;

    return (
      <div className="mt-3 flex flex-col gap-3">
        {hasQuestion && (
          <blockquote className="rounded-lg border border-c-neutral-200 bg-base-300/70 px-4 py-3 text-base italic opacity-95">
            <span className="mr-2 text-md not-italic font-semibold uppercase tracking-wide">User input:</span>
            <span className="mr-2 opacity-80">“</span>
            <RichDescription description={item.question} className="inline" />
            <span className="ml-1 opacity-80">”</span>
          </blockquote>
        )}
        {hasResponse && (
          <div className="rounded-lg border border-c-neutral-200 bg-base-100/70 px-4 py-3">
            <p className="mb-1 text-sm font-semibold uppercase tracking-wide">
              AI tool/agent response
            </p>
            <RichDescription description={item.response} className="mb-0 text-base opacity-95" />
          </div>
        )}
      </div>
    );
  };

  const copyToClipboard = (text, blockKey) => {
    navigator.clipboard.writeText(text).then(() => {
      setCopiedBlockKey(blockKey);
      setTimeout(() => setCopiedBlockKey(null), 2000);
    });
  };

  const CodePanels = ({ block, blocks, cardIndex }) => {
    const normalized = Array.isArray(blocks)
      ? blocks
      : block
      ? [block]
      : [];

    if (!normalized.length) return null;

    const toggleCodeBlock = (index) => {
      const key = `${cardIndex}-${index}`;
      setExpandedCodeBlocks((prev) => ({
        ...prev,
        [key]: !prev[key],
      }));
    };

    return (
      <>
        {normalized.map((entry, index) => {
          const resolved = typeof entry === "string" ? { code: entry } : entry;
          const code = String(resolved?.code || "");
          if (!code.trim()) return null;

          const language = String(resolved.language || "text").toLowerCase();
          const label =
            resolved.label || resolved.title || language.toUpperCase();
          const blockKey = `${cardIndex}-${index}`;
          const isExpanded = expandedCodeBlocks[blockKey] || false;

          return (
            <div
              key={`${label}-${index}`}
              className="mt-4 rounded-xl border border-c-neutral-200 bg-base-300 p-3"
            >
              <div className="mb-0 flex items-center justify-between gap-2">
                <button
                  onClick={() => toggleCodeBlock(index)}
                  className="inline-flex items-center rounded-md bg-base-200 px-2 py-1 text-sm font-semibold uppercase tracking-wide opacity-80 hover:opacity-100 transition-opacity cursor-pointer"
                >
                  <span className="mr-2">
                    {isExpanded ? "▼" : "▶"}
                  </span>
                  {label}
                </button>
                <button
                  onClick={() => copyToClipboard(code, blockKey)}
                  className="inline-flex items-center rounded-md bg-base-200 px-2 py-1 text-sm font-semibold uppercase tracking-wide opacity-80 hover:opacity-100 transition-opacity cursor-pointer"
                >
                  {copiedBlockKey === blockKey ? (
                    <>
                      <span className="mr-1">✓</span>
                      Copied
                    </>
                  ) : (
                    <>
                      <span className="mr-1">📋</span>
                      Copy
                    </>
                  )}
                </button>
              </div>
              {isExpanded && (
                  <pre className="m-0 overflow-auto whitespace-pre-wrap break-words text-sm leading-relaxed" style={{marginBottom: '0', marginTop:'15px'}}>
                    <code className={`language-${language} font-mono`}>{code}</code>
                  </pre>
              )}
            </div>
          );
        })}
      </>
    );
  };

  return (
    <div className={["bg-base-300", containerClasses].join(" ")}>
      <div className="container mx-auto px-0 pb-4 pt-4">
        <div className={["grid grid-cols-1 gap-8", colsClass].join(" ")}>
          {items.map((item, idx) => {
            const isImage = item.adornment?.type === "image";
            const isCenteredLayout = layout === "centered";
            const isQA =
              item.variant === "qa" ||
              item.type === "qa" ||
              item.question != null ||
              item.response != null;

            return (
              <div
                key={idx}
                className="card relative overflow-hidden border border-c-neutral-200 shadow-lg transition-all duration-300 hover:-translate-y-1 hover:shadow-2xl"
                style={{
                  background:
                    "radial-gradient(130% 100% at 50% -10%, color-mix(in srgb, var(--color-accent) 12%, transparent), transparent 65%), linear-gradient(180deg, color-mix(in srgb, var(--color-base-200) 92%, var(--color-base-300)), var(--color-base-200))",
                }}
              >
                {/* Image layout */}
                {isImage ? (
                  <div className="card-body gap-0 p-5 sm:p-6">
                    <div className="flex flex-col items-center md:flex-column md:items-start gap-0 w-full text-center">
                      <div className="flex items-center justify-center md:justify-center px-4 w-full">
                        <div className="figure-box relative flex items-center justify-center md:justify-start mb-2 h-[200px] w-full">
                          <div
                            className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-[220px] h-[150px] rounded-full pointer-events-none"
                            style={{
                              background:
                                "radial-gradient(ellipse at center, color-mix(in srgb, var(--color-accent) 72%, transparent) 0%, color-mix(in srgb, var(--color-primary) 38%, transparent) 45%, transparent 78%)",
                              filter: "blur(24px)",
                              opacity: 0.6,
                            }}
                          />
                          <img
                            src={item.adornment.imageUrl}
                            alt={item.adornment.alt || ""}
                            className="relative z-10 max-h-full object-contain"
                            loading="lazy"
                            decoding="async"
                          />
                        </div>
                      </div>
                      <div className="w-full text-center md:text-center">
                        <h3 className="card-title text-lg font-black mb-2 mt-0 text-center block">
                          {item.title}
                        </h3>
                        {!isQA && (
                          <RichDescription
                            description={item.description}
                            className="text-md mb-0 block opacity-90"
                          />
                        )}
                        {isQA && <QAContent item={item} />}
                        <CodePanels block={item.codeBlock} blocks={item.codeBlocks} cardIndex={idx} />
                        <CardButton button={item.button} />
                      </div>
                    </div>
                  </div>
                ) : (
                  /* Default icon/number layout */
                  <div className="card-body gap-0 p-5 sm:p-6">
                    <div
                      className={[
                        "flex gap-0 w-full",
                        isCenteredLayout
                          ? "flex-col items-center"
                          : "flex-col md:flex-row items-center md:items-start",
                      ].join(" ")}
                    >
                      {/* Left adornment */}
                      <div
                        className={[
                          "flex items-center",
                          isCenteredLayout
                            ? "justify-center w-full pr-0"
                            : "justify-center md:justify-start pr-4",
                          item.adornment?.type === "numberIcon"
                            ? isCenteredLayout
                              ? "w-full"
                              : "w-full md:w-[140px]"
                            : "w-auto",
                        ].join(" ")}
                      >
                        {item.adornment?.type === "icon" && (
                          <div
                            className={[
                              "figure-box flex items-center mb-2 h-[60px]",
                              isCenteredLayout
                                ? "justify-center w-full"
                                : "justify-center md:justify-start",
                            ].join(" ")}
                          >
                            <span
                              className={[
                                "text-6xl drop-shadow-sm color-accent",
                                normalizeToneClasses(item.adornment.iconClass || "")
                              ].join(" ")}
                            />
                          </div>
                        )}
                        {item.adornment?.type === "numberIcon" && (
                          <div className="figure-box flex items-center justify-center md:justify-start mb-2">
                            {item.adornment.numberText ? (
                              <span className="figure text-4xl font-black font-mono">
                                {item.adornment.numberText}
                              </span>
                            ) : null}
                            {item.adornment.iconClass ? (
                              <span
                                className={[
                                  "ml-2 text-2xl color-accent",
                                  normalizeToneClasses(item.adornment.iconClass)
                                ].join(" ")}
                              />
                            ) : null}
                          </div>
                        )}
                      </div>

                      {/* Content area */}
                      <div
                        className={[
                          "w-full",
                          isCenteredLayout
                            ? "text-center"
                            : "md:flex-1 text-center md:text-left",
                        ].join(" ")}
                      >
                        <h3
                          className={[
                            "card-title text-lg font-black mb-2 mt-0",
                            isCenteredLayout ? "justify-center" : "",
                          ].join(" ")}
                        >
                          {item.title}
                        </h3>
                        <div
                          className={[
                            "flex flex-col gap-2",
                            isCenteredLayout
                              ? "items-center"
                              : "md:flex-row items-center md:items-start md:justify-between",
                          ].join(" ")}
                        >
                          <div className={isCenteredLayout ? "w-full" : "w-full md:flex-1"}>
                            {!isQA && (
                              <RichDescription
                                description={item.description}
                                className={[
                                  "text-md mb-0 opacity-90",
                                  isCenteredLayout ? "mx-auto" : "",
                                ].join(" ")}
                              />
                            )}
                            {isQA && <QAContent item={item} />}
                          </div>
                          {item.copyText && (
                            <button
                              onClick={() => copyToClipboard(item.copyText, `card-${idx}`)}
                              className="inline-flex items-center shrink-0 rounded-md bg-base-200 px-2 py-1 text-sm font-semibold uppercase tracking-wide opacity-80 hover:opacity-100 transition-opacity cursor-pointer"
                            >
                              {copiedBlockKey === `card-${idx}` ? (
                                <>
                                  <span className="mr-1">✓</span>
                                  Copied
                                </>
                              ) : (
                                <>
                                  <span className="mr-1">📋</span>
                                  Copy
                                </>
                              )}
                            </button>
                          )}
                        </div>
                        {(item.dataAccess || item.useCase) && (
                          <div
                            className={[
                              "mt-3 gap-3 text-sm",
                              isCenteredLayout ? "grid grid-cols-1" : "grid grid-cols-2",
                            ].join(" ")}
                          >
                            {item.dataAccess && (
                              <div className="col-span-1">
                                <p className="font-semibold text-opacity-75 opacity-75 uppercase tracking-wide mb-1">Data access</p>
                                <RichDescription
                                  description={item.dataAccess}
                                  className="text-sm opacity-90"
                                />
                              </div>
                            )}
                            {item.useCase && (
                              <div className="col-span-1">
                                <p className="font-semibold text-opacity-75 opacity-75 uppercase tracking-wide mb-1">Use case</p>
                                <RichDescription
                                  description={item.useCase}
                                  className="text-sm opacity-90"
                                />
                              </div>
                            )}
                          </div>
                        )}
                        <CodePanels block={item.codeBlock} blocks={item.codeBlocks} cardIndex={idx} />
                        <CardButton button={item.button} centered={isCenteredLayout} />
                      </div>
                    </div>
                  </div>
                )}
              </div>
            );
          })}

          {items.length === 0 && (
            <div className="text-center opacity-70 py-8">
              <span>No cards to display</span>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

  gridColsMd={2}
  items={[
    {
      adornment: {
        type: "image",
        imageUrl: "https://files.readme.io/e840c5bd2a758d2a2afaff923164aee9ef86d78a3b327c97c1b31dbc40026aa5-learn.png",
        alt: "Learn GraphQL"
      },
      title: "Getting to know GraphQL",
      description: <>47% of developers reported using <span className="text-accent font-semibold">GraphQL</span> in 2023.</>,
      button: { text: "Learn more", href: "#", target: "_self" }
    },
    {
      adornment: { type: "icon", iconClass: "fa-solid fa-bolt text-warning" },
      title: "Performance Boost",
      description: <>Teams report 30-50% faster cycles by fetching only the data they need.</>,
      button: { text: "See examples", href: "#examples" }
    }
  ]}
/>
