  return (
    <li className="relative grid grid-cols-[2.5rem_1fr] items-start font-inherit">
      {/* Rail */}
      <div className="relative flex justify-center self-stretch">
        {/* Step circle */}
        <span
          className="
            relative z-10 flex h-8 w-8 items-center justify-center

            text-sm font-semibold text-[#4155FF] dark:bg-[#4155FF]/15
          "
        >
          {index}
        </span>

        {/* Connector — from below this circle to above the next circle */}
        {!isLast && (
          <span
            aria-hidden

              bg-[#4155FF]/30 dark:bg-[#4155FF]/40

          />
        )}
      </div>

      {/* Content */}
      <div className="pt-0.5 font-inherit">
        {children}

    </li>
  )
}
